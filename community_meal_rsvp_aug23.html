function doPost(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const data = e.parameter && e.parameter.data ? JSON.parse(e.parameter.data) : JSON.parse(e.postData.contents); sheet.appendRow([ data.timestamp, data.name, data.email, data.attending, data.headcount || '', data.allergy, data.allergyDetail || '', data.items || '' ]); if (data.attending === 'yes' && data.email) { let body = 'Hi ' + data.name + ',\n\nThanks for your RSVP! We look forward to seeing you'; const guests = parseInt(data.headcount) || 1; if (guests > 1) body += ' and your ' + (guests - 1) + ' guest' + (guests > 2 ? 's' : ''); body += ' at our Community Meal on Sunday 23rd August.\n\n'; if (data.items) { body += 'You have signed up to bring:\n'; data.items.split(',').forEach(item => { body += ' - ' + item.trim() + '\n'; }); body += '\n'; } body += 'If you have any questions please don\'t hesitate to get in touch.\n\nGod bless,\nThe Fields Church'; MailApp.sendEmail({ to: data.email, subject: 'Community Meal RSVP Confirmation - Sunday 23rd August', body: body }); } return ContentService.createTextOutput('OK').setMimeType(ContentService.MimeType.TEXT); } function doGet(e) { const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); const rows = sheet.getDataRange().getValues(); const claimed = { 'Bag of grated cheese': 0, 'Tub of sour cream': 0, 'Jar of jalapenos': 0, 'Tub of butter': 0, 'Tin of corn': 0, 'Container of sliced cooked mushrooms': 0 }; for (let i = 1; i < rows.length; i++) { if (rows[i][7]) { const items = rows[i][7].split(','); items.forEach(entry => { const match = entry.trim().match(/^(\d+)x (.+)$/); if (match) { const qty = parseInt(match[1]); const name = match[2].trim(); if (claimed[name] !== undefined) claimed[name] += qty; } }); } } const json = JSON.stringify(claimed); const output = ContentService.createTextOutput(json); output.setMimeType(ContentService.MimeType.JSON); return output; }
